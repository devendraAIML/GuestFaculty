import javafx.fxml.FXML;
import javafx.scene.control.*;
import javafx.scene.layout.GridPane;
import javafx.scene.layout.VBox;
import javafx.scene.layout.AnchorPane;
import javafx.scene.control.Alert.AlertType;
import javafx.scene.control.ButtonType;

import javafx.fxml.FXMLLoader;
import javafx.scene.Parent;
import javafx.scene.Scene;
import javafx.stage.Stage;
import javafx.scene.Node;
import javafx.event.ActionEvent;

import java.sql.*;
import java.util.*;

public class AdminController {

    @FXML private AnchorPane mainContent;
    @FXML private GridPane requestPane;
    @FXML private VBox viewDetailsBox;

    @FXML private VBox feedbackButtonsBox;
    @FXML private VBox summaryButtonsBox;

    @FXML private TableView<?> sessionDetailsTable;
    @FXML private TableView<?> ratingTable;
    @FXML private TableView<?> feedbackTable;
    @FXML private TableView<?> summaryTable;

    @FXML private ComboBox<String> userTypeBox;
    @FXML private TextField usernameField;
    @FXML private TextField passwordField;
    @FXML private TextField contactField;

    @FXML private TableColumn<?, ?> colFacultyName;
    @FXML private TableColumn<?, ?> colSessionName;
    @FXML private TableColumn<?, ?> colSessionDuration;
    @FXML private TableColumn<?, ?> colSessionDate;

    // ================= DATABASE =================
    private final String DB_URL = "jdbc:oracle:thin:@localhost:1521:XE";
    private final String DB_USER = "system";
    private final String DB_PASS = "Devendra";

    // ================= REQUEST STORAGE =================
    private List<Request> requests = new ArrayList<>();
    private int currentIndex = 0;

    // ================= INITIALIZE =================
    @FXML
    public void initialize() {

        userTypeBox.getItems().addAll("Faculty", "Student");

        requestPane.setVisible(false);
        viewDetailsBox.setVisible(false);
        sessionDetailsTable.setVisible(false);

        feedbackButtonsBox.setVisible(false);
        summaryButtonsBox.setVisible(false);

        ratingTable.setVisible(false);
        feedbackTable.setVisible(false);
        summaryTable.setVisible(false);

        loadRequestsFromDB();
    }

    // ================= LOAD REQUESTS =================
    private void loadRequestsFromDB() {

        requests.clear();

        try (Connection conn = DriverManager.getConnection(DB_URL, DB_USER, DB_PASS)) {

            // STUDENT REQUESTS
            PreparedStatement ps = conn.prepareStatement("SELECT * FROM student_login_requests");
            ResultSet rs = ps.executeQuery();

            while (rs.next()) {
                requests.add(new Request(
                        "Student",
                        rs.getInt("request_id"),
                        rs.getString("name"),
                        rs.getString("username"),
                        rs.getString("password"),
                        null,
                        rs.getString("registernumber"),
                        rs.getString("branch"),
                        rs.getString("section")
                ));
            }

            // FACULTY REQUESTS
            ps = conn.prepareStatement("SELECT * FROM faculty_login_requests");
            rs = ps.executeQuery();

            while (rs.next()) {
                requests.add(new Request(
                        "Faculty",
                        rs.getInt("request_id"),
                        rs.getString("name"),
                        rs.getString("username"),
                        rs.getString("password"),
                        rs.getString("gmail"),
                        null,
                        null,
                        null
                ));
            }

        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    // ================= SHOW REQUEST PANEL =================
    @FXML
    private void showRequests() {

        requestPane.setVisible(true);
        viewDetailsBox.setVisible(false);
        sessionDetailsTable.setVisible(false);

        feedbackButtonsBox.setVisible(false);
        summaryButtonsBox.setVisible(false);

        ratingTable.setVisible(false);
        feedbackTable.setVisible(false);
        summaryTable.setVisible(false);

        currentIndex = 0;
        loadNextRequest();
    }

    // ================= LOAD NEXT REQUEST =================
    private void loadNextRequest() {

        if (currentIndex >= requests.size()) {
            showAlert(AlertType.INFORMATION, "No Requests", "No requests available.");
            clearFields();
            return;
        }

        Request r = requests.get(currentIndex);

        userTypeBox.setValue(r.userType);
        usernameField.setText(r.username);
        passwordField.setText(r.password);

        if (r.userType.equals("Student")) {
            contactField.setText(
                    "RegNo: " + r.regNo +
                    " | Branch: " + r.branch +
                    " | Section: " + r.section +
                    " | Name: " + r.name
            );
        } else {
            contactField.setText(
                    "Email: " + r.email +
                    " | Name: " + r.name
            );
        }
    }

    // ================= ACCEPT REQUEST =================
    @FXML
    private void acceptRequest() {

        if (currentIndex >= requests.size()) return;

        Request r = requests.get(currentIndex);

        try (Connection conn = DriverManager.getConnection(DB_URL, DB_USER, DB_PASS)) {

            conn.setAutoCommit(false);

            if (r.userType.equals("Student")) {

                PreparedStatement ps = conn.prepareStatement(
                        "INSERT INTO student_login(name, registernumber, branch, section, username, password) VALUES(?,?,?,?,?,?)");

                ps.setString(1, r.name);
                ps.setString(2, r.regNo);
                ps.setString(3, r.branch);
                ps.setString(4, r.section);
                ps.setString(5, r.username);
                ps.setString(6, r.password);
                ps.executeUpdate();

                ps = conn.prepareStatement("DELETE FROM student_login_requests WHERE request_id=?");
                ps.setInt(1, r.requestId);
                ps.executeUpdate();

            } else {

                PreparedStatement ps = conn.prepareStatement(
                        "INSERT INTO faculty_login(name, username, gmail, password) VALUES(?,?,?,?)");

                ps.setString(1, r.name);
                ps.setString(2, r.username);
                ps.setString(3, r.email);
                ps.setString(4, r.password);
                ps.executeUpdate();

                ps = conn.prepareStatement("DELETE FROM faculty_login_requests WHERE request_id=?");
                ps.setInt(1, r.requestId);
                ps.executeUpdate();
            }

            conn.commit();

            showAlert(AlertType.INFORMATION, "Success", "Account Created Successfully");

            currentIndex++;
            loadNextRequest();

        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    // ================= REJECT REQUEST =================
    @FXML
    private void rejectRequest() {

        if (currentIndex >= requests.size()) return;

        Request r = requests.get(currentIndex);

        if (!confirmDialog("Reject", "Are you sure?")) return;

        try (Connection conn = DriverManager.getConnection(DB_URL, DB_USER, DB_PASS)) {

            PreparedStatement ps;

            if (r.userType.equals("Student")) {
                ps = conn.prepareStatement("DELETE FROM student_login_requests WHERE request_id=?");
            } else {
                ps = conn.prepareStatement("DELETE FROM faculty_login_requests WHERE request_id=?");
            }

            ps.setInt(1, r.requestId);
            ps.executeUpdate();

            showAlert(AlertType.WARNING, "Rejected", "Request Deleted");

            currentIndex++;
            loadNextRequest();

        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    // ================= OTHER UI METHODS (UNCHANGED) =================
    @FXML
    private void showSession() {
        requestPane.setVisible(false);
        viewDetailsBox.setVisible(true);
    }

    @FXML
    private void showFeedback() {
        feedbackButtonsBox.setVisible(true);
    }

    @FXML
    private void showSummary() {
        summaryButtonsBox.setVisible(true);
    }

    @FXML
    private void showSessionDetails() {
        sessionDetailsTable.setVisible(true);
    }

    @FXML
    private void showRatingTable() {
        ratingTable.setVisible(true);
        feedbackTable.setVisible(false);
    }

    @FXML
    private void showFeedbackTable() {
        feedbackTable.setVisible(true);
        ratingTable.setVisible(false);
    }

    @FXML
    private void generateSummary() {
        summaryTable.setVisible(true);
    }

    @FXML
    private void logout(ActionEvent event) {
        showAlert(AlertType.INFORMATION, "Logout", "Logged out successfully");

        try {
            Parent root = FXMLLoader.load(getClass().getResource("Login.fxml"));
            Stage stage = (Stage)((Node)event.getSource()).getScene().getWindow();
            stage.setScene(new Scene(root));
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    // ================= HELPERS =================
    private void clearFields() {
        userTypeBox.setValue(null);
        usernameField.clear();
        passwordField.clear();
        contactField.clear();
    }

    private void showAlert(AlertType type, String title, String msg) {
        Alert alert = new Alert(type);
        alert.setTitle(title);
        alert.setHeaderText(null);
        alert.setContentText(msg);
        alert.showAndWait();
    }

    private boolean confirmDialog(String title, String msg) {
        Alert alert = new Alert(AlertType.CONFIRMATION);
        alert.setTitle(title);
        alert.setHeaderText(null);
        alert.setContentText(msg);
        Optional<ButtonType> result = alert.showAndWait();
        return result.isPresent() && result.get() == ButtonType.OK;
    }

    // ================= REQUEST CLASS =================
    private static class Request {
        String userType;
        int requestId;
        String name, username, password;
        String email;
        String regNo, branch, section;

        Request(String userType, int requestId, String name, String username, String password,
                String email, String regNo, String branch, String section) {
            this.userType = userType;
            this.requestId = requestId;
            this.name = name;
            this.username = username;
            this.password = password;
            this.email = email;
            this.regNo = regNo;
            this.branch = branch;
            this.section = section;
        }
    }
}
